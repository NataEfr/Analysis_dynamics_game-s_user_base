##### **2. Анализ игровых сессий.**



###### 2.1 Суммарное количество игровых сессий за все время

select count(\*) as cnt\_all

from skygame.game\_sessions

###### 

###### 2.2 Определим суммарное кол-во сессий дольше 5 минут



select count(\*) as cnt\_all

&#x20;       ,sum(case when end\_session- start\_session > interval '5 minute' then 1 else 0 end) as cnt\_signif

from skygame.game\_sessions





###### 2.3 Динамика по месяцам "суммарное кол-во сессий":

###### &#x20;   -- суммарное кол-во сессий дольше 5 минут

###### &#x20;   -- доля сессий дольше 5 минут





select date\_trunc('month', start\_session) as mm

&#x20;       ,count(\*) as cnt\_all

&#x20;       ,sum(case when end\_session- start\_session > interval '5 minute' then 1 else 0 end) as cnt\_signif

&#x20;       , sum(case when end\_session- start\_session > interval '5 minute' then 1 else 0 end)::float/count(\*) as share\_signif

from skygame.game\_sessions

&#x20;   group by mm

&#x20;   order by mm

&#x20;

\-- динамика средней длительности одной игровой сессии по месяцам

select date\_trunc('month', start\_session) as mm

&#x20;      , avg(extract(epoch from end\_session - start\_session))/(60\*60) as avg\_len

from skygame.game\_sessions

where end\_session - start\_session > interval '5 minute'

&#x20;   group by mm

&#x20;   order by mm

&#x20;

&#x20;

\-- Динамика доли длинных сессий среди всех сессий. Граница 1 час



select date\_trunc('month', start\_session) as mm

&#x20;       , sum(case when end\_session- start\_session > interval'1 hour' then 1 else 0 end):: float/count(\*) as share\_long

&#x20;      from skygame.game\_sessions

where end\_session - start\_session > interval '5 minute'

&#x20;   group by mm

&#x20;   order by mm

select    count(distinct id\_user) as cnt\_user

&#x20;       , sum(ref\_reg) / count(\*) as share\_reg

from skygame.referral

###### 3.2 Выведите топ-50 пользователей, у которых больше всего приглашений.



select    id\_user

&#x20;       , count(\*) as cnt\_ref

from skygame.referral

group by id\_user

order by cnt\_ref desc

limit 50



\------------------------------------------------------------

###### 3.3 Выведите тех пользователей, которые сделали больше 5 приглашений и у которых минимум половина приглашенных зарегистрировались.



select    id\_user

&#x20;       , count(\*) as cnt\_ref

&#x20;       , sum(ref\_reg)/count(\*) as share\_reg

from skygame.referral

group by id\_user

having count(\*) > 5

&#x20; and sum(ref\_reg)/count(\*) >= 0.5



\-------------------------------------------------------------
